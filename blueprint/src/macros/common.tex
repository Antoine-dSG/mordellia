% In this file you should put all LaTeX macros and settings to be used both by

% The theorem-like environments defined below are those that appear by default
% in the dependency graph. See the README of leanblueprint if you need help to
% customize this. 
% The configuration below use the theorem counter for all those environments
% (this is what the [theorem] arguments mean) and never resets it.
% If you want for instance to number them within chapters then you can add
% [chapter] at the end of the next line.
\newtheorem{theorem}{Theorem}
\newtheorem{proposition}[theorem]{Proposition}
\newtheorem{conjecture}[theorem]{Conjecture}
\newtheorem{lemma}[theorem]{Lemma}
\newtheorem{corollary}[theorem]{Corollary}

\theoremstyle{definition}
\newtheorem{definition}[theorem]{Definition}

\setlength{\parindent}{0pt}
\setlength{\parskip}{0.25\baselineskip}
\renewcommand{\arraystretch}{1.15}
\providecommand{\allowdisplaybreaks}[1][]{}
\allowdisplaybreaks
\raggedbottom
\sloppy

\providecommand{\Needspace}[1]{}

\newcommand{\PrimaryEquation}[6]{%
  \par\Needspace{5\baselineskip}\smallskip
  \noindent\hypertarget{eq:#1}{}\hypertarget{eq:#2}{}%
  \textbf{#1}\quad{\footnotesize\textbf{#2}\quad\textsc{#4}\quad\footnotesize\textit{#5.}}\par\nopagebreak
  {%
    \small
    \setlength{\abovedisplayskip}{3pt}%
    \setlength{\belowdisplayskip}{3pt}%
    \[\displaystyle #3\]
  }%
  {#6}\par
}

\newcommand{\EquationBacklink}[4]{%
  \par\Needspace{2\baselineskip}\smallskip
  \noindent\hyperlink{eq:#1}{\textbf{#1}}\quad
  \hyperlink{eq:#1}{See the full entry in Chapter~#2}.%
  \quad{\footnotesize Local occurrence: Mordell #3; \textsc{#4}.}\par
}

\newcommand{\ChapterHeading}[4]{%
  \Needspace{7\baselineskip}%
  \phantomsection\hypertarget{chapter:#1}{}%
  \section*{Chapter #1. #2}%
  \textit{Printed pages #3--#4.}\par
}
