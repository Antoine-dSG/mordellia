\begin{proposition}
    The equation
\[
    z^2=(a_1x^2+b_1y^2)(a_2x^2+b_2y^2)(a_3x^2+b_3y^2)
\]
    has no integer solutions except $(0,0,0)$ if all the following conditions hold:
\[
    a_1+a_2+a_3\equiv-1\pmod4,\qquad b_1+b_2+b_3\equiv0\pmod4,
\]
    only one of the $b_i$ is even, none of the triples
\[
    (a_1,a_2,a_3),\qquad (a_1+b_1,a_2+b_2,a_3+b_3)
\]
    is congruent modulo $4$ to $(1,1,1)$ or to a permutation of $(0,1,2)$, and $a_2b_3-a_3b_2$, $a_3b_1-a_1b_3$, $a_1b_2-a_2b_1$ have no common divisor and have all odd factors congruent to $1\pmod4$.
\end{proposition}
