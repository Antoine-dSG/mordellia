\begin{proposition}
    The equation
\[
    x^4+x^2y^2+y^4=z^2,\qquad (x,y)=1,
\]
    has only the solutions $x^2=1$, $y=0$, and $x=0$, $y^2=1$.
\end{proposition}

\begin{proposition}
    The equation
\[
    x^4-x^2y^2+y^4=z^2,\qquad (x,y)=1,
\]
    has only the solutions $x^2=1$, $y=0$; $x=0$, $y^2=1$; and $x^2=y^2=1$.
\end{proposition}
