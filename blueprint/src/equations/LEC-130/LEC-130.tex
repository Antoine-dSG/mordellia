\begin{proposition}
    The equation
\[
    x^4-Dy^4=1
\]
    where $D$ is a positive non-square integer, has at most one integer solution. If this is $(x_1,y_1)$ and $\varepsilon$ is the fundamental unit in $\mathbb{Q}(\sqrt{D})$ with $N(\varepsilon)=1$, then
\[
    x_1^2+y_1^2\sqrt{D}=\varepsilon\quad\text{or}\quad\varepsilon^2,
\]
    the latter occurring in only finitely many cases. If $N(\varepsilon)=-1$, there is no solution except when $D=5$.
\end{proposition}

\begin{proposition}
    The equations
\[
    x^4-Dy^4=\pm1
\]
    have at most one solution in positive integers.
\end{proposition}
