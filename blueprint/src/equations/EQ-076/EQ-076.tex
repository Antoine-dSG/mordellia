\begin{proposition}
    Let $f_s(x_s,y_s,z_s)$, $s=1,2,3$, be cubic homogeneous polynomials such that the only solution of $f_s(x_s,y_s,z_s)\equiv0\pmod p$ is $x_s\equiv y_s\equiv z_s\equiv0\pmod p$. Then
\[
    \begin{aligned}&f_1(x_1,y_1,z_1)+pf_2(x_2,y_2,z_2)+p^2f_3(x_3,y_3,z_3)\\&\qquad{}+p^3\sum_{\ell,m,n}a_{\ell mn}x_\ell y_mz_n=0\end{aligned}
\]
    has only the trivial solution $x_s=y_s=z_s=0$ for $s=1,2,3$.
\end{proposition}
