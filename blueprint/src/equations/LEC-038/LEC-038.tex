\begin{proposition}
    The congruence
\[
    ax^3+3bx^2y+3cxy^2+dy^3\equiv z^3\pmod 9
\]
    has only the trivial solution $x\equiv y\equiv z\equiv0\pmod3$ if
\[
    a,d\not\equiv0,\pm1\pmod9
\]
    and
\[
    \frac{a-a^3}{3}+c-ad^2
    \pm\left(b-a^2d+\frac{d-d^3}{3}\right)
    \not\equiv0\pmod3.
\]
\end{proposition}
