\begin{proposition}
Suppose that $m>0$, $(ab,m)=1$, and that
\[
    ak^2+b\equiv0\pmod m
\]
is solvable. Then integers $x,y$, not both zero, exist such that
\[
    ax^2+by^2=Mm
\]
for some integer $M$ with
\[
    M<2\sqrt{ab}\quad\text{if }a>0,\ b>0,
\]
and
\[
    |M|<\sqrt{|ab|}\quad\text{if }ab<0.
\]
\end{proposition}

\begin{proposition}
The equations
\[
    ax^2+by^2=\ell p,\qquad ax_1^2+by_1^2=mp,
\]
with $(x,y)=1$ and $(x_1,y_1)=1$, cannot both be solvable if
\[
    a>0,\qquad b>0,\qquad (ab,p)=1,\qquad \ell\ne m,\qquad \ell m<ab.
\]
\end{proposition}
