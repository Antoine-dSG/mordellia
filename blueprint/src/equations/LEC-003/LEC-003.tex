\begin{proposition}
The $m$ simultaneous equations
\[
    L_r=a_{r1}x_1+a_{r2}x_2+\cdots+a_{rn}x_n=0
    \quad(r=1,2,\ldots,m),\quad m<n,
\]
where the $a$'s are integers, have an integer solution $\mathbf{x}\ne0$ for which
\[
    |x_s|\le (A_1A_2\cdots A_m)^{1/(n-m)}
    \quad(s=1,2,\ldots,n),
\]
where
\[
    A_r=|a_{r1}|+|a_{r2}|+\cdots+|a_{rn}|.
\]
\end{proposition}
