\begin{proposition}
    The equation
\[
    \sum_{r,s}a_{rs}x_rx_s=0
\]
    is solvable in integers if and only if $f(\mathbf{x})\equiv0\pmod{p^r}$ is solvable for all primes $p$ and integers $r\ge1$ with $(x_1,\ldots,x_n,p)=1$, and $f(\mathbf{x})=0$ is solvable in the real field.
\end{proposition}
