\begin{proposition}
    If $f(x,y)=ax^3+bx^2y+cxy^2+dy^3$ is irreducible over $\mathbb{Q}$, then
\[
    f(x,y)=m
\]
    has only finitely many integer solutions.
\end{proposition}

\begin{proposition}
    If the discriminant $D$ of $f(x,y)=ax^3+bx^2y+cxy^2+dy^3$ is negative, then
\[
    f(x,y)=1
\]
    has at most three integer solutions, except when
\[
    f\sim(1,0,1,1)\quad\text{or}\quad f\sim(1,-1,1,1),
\]
    when there are exactly four solutions, and except when
\[
    f\sim(1,0,-1,1),
\]
    when there are exactly five solutions.
\end{proposition}

\begin{proposition}
    If the discriminant of $f(x,y)=ax^3+bx^2y+cxy^2+dy^3$ is sufficiently great, then
\[
    f(x,y)=1
\]
    has at most eighteen integer solutions.
\end{proposition}
