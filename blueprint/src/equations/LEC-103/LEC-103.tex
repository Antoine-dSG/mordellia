\begin{proposition}
    If $p\ne0$, $p\mid(\ell,m)$, and
\[
    ax^2+bxy+cy^2-\frac{(\ell x+my)^2}{4p^2}=\pm2p
\]
    has infinitely many integer solutions, then
\[
    \begin{aligned}z^2={}&p^2+\ell x+my+ax^2+bxy+cy^2\\&{}+Ax^3+Bx^2y+Cxy^2+Dy^3\end{aligned}
\]
    has infinitely many integer solutions.
\end{proposition}
