\begin{proposition}
    If the right-hand side has no squared linear factors, then
\[
    y^2=ax^4+bx^3+cx^2+dx+e
\]
    has only finitely many integer solutions.
\end{proposition}

\begin{proposition}
    If
\[
    y^2=ax^4+bx^3+cx^2+dx+e
\]
    with rational coefficients has a solution in an algebraic number field of odd degree at least $3$, then it has a rational solution.
\end{proposition}
