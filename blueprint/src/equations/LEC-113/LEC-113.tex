\begin{proposition}
    All integer solutions of
\[
    x^2+ky^2=z^3,\qquad (x,z)=1,
\]
    are obtained from binary cubic forms with binomial integer coefficients and discriminant $4k$ by
\[
    x=G(u,v),\qquad y=f(u,v),\qquad z=H(u,v),
\]
    where
\[
    G(u,v)^2+kf(u,v)^2=H(u,v)^3.
\]
\end{proposition}
