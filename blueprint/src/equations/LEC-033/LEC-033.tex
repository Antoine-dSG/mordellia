\begin{proposition}
Let $u_0=0$, $u_1=1$, and $u_{n+2}=u_{n+1}+u_n$. Then
\[
    u_n=x^2
\]
only when
\[
    n=0,\ \pm1,\ 2,\ 12,\qquad \pm x=0,\ 1,\ 1,\ 12.
\]
\end{proposition}

\begin{proposition}
Let $v_0=2$, $v_1=1$, and $v_{n+2}=v_{n+1}+v_n$. Then
\[
    v_n=x^2
\]
only when
\[
    n=1,\ 3,\qquad \pm x=1,\ 2.
\]
\end{proposition}
