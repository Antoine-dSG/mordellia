\begin{proposition}
The integer solutions of
\[
    x^2+y^2=2z^2,\qquad (x,y)=1,\qquad z>0,
\]
have $z$ odd and $x,y$ odd. Writing
\[
    x=X+Y,\qquad y=X-Y,\qquad (X,Y)=1,
\]
one has
\[
    X^2+Y^2=z^2,
\]
and hence either
\[
    X=a^2-b^2,\qquad Y=2ab,
\]
or
\[
    X=2ab,\qquad Y=a^2-b^2.
\]
\end{proposition}
