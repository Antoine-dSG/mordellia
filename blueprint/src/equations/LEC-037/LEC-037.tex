\begin{proposition}
    The equation
\[
    ax^3+3bx^2y+3cxy^2+dy^3=z^3
\]
    has no integer solutions if
\[
    a\equiv d\equiv4\pmod9,\qquad b\equiv0\pmod3,\qquad c\equiv\pm1\pmod3.
\]
    The equation
\[
    ax^3+3bx^2y+3cxy^2+dy^3=1
\]
    has no rational solutions under the same hypotheses.
\end{proposition}
