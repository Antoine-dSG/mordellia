\begin{proposition}
    The equation
\[
    \begin{aligned}&(a_1x_1^2+a_2x_2^2+a_3x_3^2+a_4x_4^2)(b_1x_1^2+b_2x_2^2+b_3x_3^2+b_4x_4^2)\\&\qquad=2k(c_1x_1^2+c_2x_2^2+c_3x_3^2+c_4x_4^2)(d_1x_1^2+d_2x_2^2+d_3x_3^2+d_4x_4^2)\end{aligned}
\]
    has only the trivial solution $x_1=x_2=x_3=x_4=0$ if the $a_i,b_i,c_i,d_i$ are all odd and
\[
    a_1\equiv a_2\equiv a_3\equiv a_4\pmod8,\qquad b_1\equiv b_2\equiv b_3\equiv b_4\pmod8,
\]
\[
    (c_1+c_2+c_3+c_4)(d_1+d_2+d_3+d_4)\equiv0\pmod{16}.
\]
\end{proposition}
