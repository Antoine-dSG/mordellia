\begin{proposition}
    Let $\rho^2+\rho+1=0$. Let $a=p$ or $p^2$, where $p\equiv2,5\pmod9$ is a prime, and let $\varepsilon$ be a unit in $\mathbb{Q}(\rho)$. Then
\[
    x^3+y^3+\varepsilon az^3=0
\]
    has no solutions in $\mathbb{Q}(\rho)$ except
\[
    z=0,\qquad x=-y,-\rho y,-\rho^2y,
\]
    unless $a=2$, when there are also the solutions
\[
    x^3=y^3=-\varepsilon z^3,\qquad \varepsilon=\pm1.
\]
\end{proposition}

\begin{proposition}
    Let $p\equiv5\pmod{18}$ and $q\equiv11\pmod{18}$ be primes. The equation
\[
    x^3+y^3=az^3
\]
    has no rational solutions except $x+y=0$, $z=0$, when
\[
    a=p,2p,9p,p^2,9p^2,4p^2,pq,p_1p_2^2,
\]
    or when
\[
    a=q,4q,9q,2q^2,q^2,9q^2,q_1q_2^2,p^2q^2,
\]
    where $p_1,p_2\equiv5\pmod{18}$ and $q_1,q_2\equiv11\pmod{18}$ are primes.
\end{proposition}
