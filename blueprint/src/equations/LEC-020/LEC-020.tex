\begin{proposition}
    Let $D$ be a positive integer which is not a perfect square. Then
\[
    y^2-Dx^2=1
\]
    has an infinity of integer solutions.
\end{proposition}

\begin{proposition}
Let $(x,y)=(U,T)$ be the solution of
\[
    y^2-Dx^2=1
\]
with least positive $x$, where $T>0$ and $U>0$. All solutions are given by
\[
    y+x\sqrt{D}=\pm(T+U\sqrt{D})^n,
\]
where $n$ is an integer.
\end{proposition}

\begin{proposition}
A necessary condition for
\[
    y^2-Dx^2=-1
\]
to be solvable is that $D$ is not divisible by $4$ or by any prime
$\equiv3\pmod4$. The equation is solvable when $D=p$ is a prime
$\equiv1\pmod4$.
\end{proposition}
