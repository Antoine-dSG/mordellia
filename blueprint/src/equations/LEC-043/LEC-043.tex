\begin{proposition}
    Let $a,b,c$ be square-free integers, relatively prime in pairs, with $abc\ne0$, and let $d$ be an integer. Then the curve
\[
    ax^3+by^3+cz^3+dxyz=0
\]
    when at most one of $a,b,c$ is $\pm1$, has either no rational points or infinitely many rational points.
\end{proposition}

\begin{proposition}
    When $a=b=1$ and $c\ne\pm1$, the curve
\[
    x^3+y^3+cz^3+dxyz=0
\]
    has either one, two, or infinitely many integer solutions. Only two solutions may be possible when $d=-(c\pm2)$ or $d=-(4c\pm1)$.
\end{proposition}

\begin{proposition}
    When $a=b=c=1$ and $d\ne1,-3,-5$, the curve
\[
    x^3+y^3+z^3+dxyz=0
\]
    has either three rational points or infinitely many rational points.
\end{proposition}
