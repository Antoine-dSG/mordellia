\begin{proposition}
If
\[
    k=ab(a+b)(a-b),
\]
then a rational solution of
\[
    y^2=x^2+k,\qquad z^2=x^2-k
\]
is given by
\[
    2x=a^2+b^2,\qquad 2y=a^2+2ab-b^2,\qquad
    2z=a^2-2ab-b^2.
\]
\end{proposition}

\begin{proposition}
For $k=5$, a rational solution is given by
\[
    x=\frac{41}{12},\qquad y=\frac{49}{12},\qquad z=\frac{31}{12}.
\]
All the solutions can be derived from the first by rational operations.
\end{proposition}
