\begin{proposition}
    The equation
\[
    y^2=x^3+k
\]
    has only finitely many integer solutions.
\end{proposition}

\begin{proposition}
    Let $k$ be free from sixth-power prime factors. If
\[
    y^2=x^3+k
\]
    has a rational solution $(x,y)=(p,q)$ with $pq\ne0$, then it has infinitely many rational solutions unless $k=-432$, when the only rational solution is $x=12$, $y=\pm36$, or $k=1$, when the only rational solutions have $x=0,-1,2$.
\end{proposition}
