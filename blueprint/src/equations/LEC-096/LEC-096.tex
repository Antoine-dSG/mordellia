\begin{proposition}
    If a non-trivial solution of
\[
    x^2+y^2+z^2-axyz=b
\]
    exists, it can be derived from a fundamental solution by finitely many elementary operations.
\end{proposition}

\begin{proposition}
    If a non-trivial solution of
\[
    x^2+y^2+z^2-axyz=b
\]
    exists, then there are infinitely many solutions except when $a=1$ and $b=2$.
\end{proposition}
