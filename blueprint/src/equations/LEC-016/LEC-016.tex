\begin{proposition}
The general integer solution of
\[
    x_1^2+x_2^2+\cdots+x_n^2=x^2,\qquad (x_1,x_2,\ldots,x_n)=1,\qquad x>0,
\]
is given by
\[
    dx_r=2X_rX_n\quad(r=1,2,\ldots,n-1),
\]
\[
    dx_n=X_n^2-X_1^2-\cdots-X_{n-1}^2,\qquad
    dx=X_n^2+X_1^2+\cdots+X_{n-1}^2,
\]
where the $X$ are arbitrary integers with $(X_1,\ldots,X_n)=1$ and $d>0$ is
taken so that $(x_1,x_2,\ldots,x_n)=1$.
\end{proposition}
