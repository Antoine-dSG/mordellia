\begin{proposition}
    The equation
\[
    y^2=Dx^4+1
\]
    has no integer solutions except $x=0$ if
\[
    D\not\equiv0,-1,3,8\pmod{16}
\]
    and $D$ has no factorization $D=pq$, $(p,q)=1$, with $p$ odd, $p>1$, and
\[
    p\equiv\pm1,\ q\pm1,\ \text{or}\ 4q\pm1\pmod{16}.
\]
\end{proposition}

\begin{proposition}
    If $D>0$ is not a square, then
\[
    y^2=Dx^4+1
\]
    has at most two solutions in positive integers.
\end{proposition}

\begin{proposition}
    If the fundamental unit of $\mathbb{Q}(\sqrt{d})$ is not also the fundamental unit of $\mathbb{Z}[\sqrt{d}]$, then
\[
    y^2=dx^4-1
\]
    has at most two possible solutions, and these can be found by a finite algorithm.
\end{proposition}
