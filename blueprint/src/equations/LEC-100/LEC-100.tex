\begin{proposition}
    The equation
\[
    x^3+y^3+z^3=n
\]
    has no integer solutions when $n\equiv\pm4\pmod9$.
\end{proposition}

\begin{proposition}
    There are infinitely many integers $n\not\equiv\pm4\pmod9$ for which
\[
    x^3+y^3+z^3=n
\]
    has no solutions in positive integers.
\end{proposition}
