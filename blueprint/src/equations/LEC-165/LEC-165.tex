\begin{proposition}
    Let $f(x,y)$ be a polynomial of degree $n$ with integer coefficients which is irreducible over $\mathbb{Q}$. If the homogeneous part $f_n(x,y)$ of degree $n$ is reducible over $\mathbb{Q}$ and is not a constant multiple of a power of an irreducible polynomial, then
\[
    f(x,y)=0
\]
    has only finitely many integer solutions.
\end{proposition}

\begin{proposition}
    The equation
\[
    f(x,y)=0
\]
    has only finitely many integer solutions if it is not identically satisfied by a parametrisation
\[
    x=A/L^m,\qquad y=B/L^m,
\]
    where $A,B,L$ are polynomials in $t$ and $L$ is linear, or by a parametrisation
\[
    x=C/Q^m,\qquad y=D/Q^m,
\]
    where $C,D,Q$ are polynomials in $t$ and $Q$ is an indefinite quadratic polynomial.
\end{proposition}
