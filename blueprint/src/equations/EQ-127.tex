\begin{proposition}\label{eq:127}
    Let
\[
    f(\mathbf{x})=x_1^4+x_2^4+x_3^4-x_2^2x_3^2-x_3^2x_1^2-x_1^2x_2^2-x_1x_2x_3(x_1+x_2+x_3).
\]
    Then
\[
    \begin{aligned}f(\mathbf{x})+f(\mathbf{y})+f(\mathbf{z})-4\{f(\mathbf{u})+f(\mathbf{v})+f(\mathbf{w})\}&=0,\\f(\mathbf{x})&=x_1^4+x_2^4+x_3^4-x_2^2x_3^2-x_3^2x_1^2-x_1^2x_2^2\\&\qquad-x_1x_2x_3(x_1+x_2+x_3)\end{aligned}
\]
    has only the trivial zero solution.
\end{proposition}
