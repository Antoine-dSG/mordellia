\begin{proposition}
    If a rational point $P=(x_1,x_2,x_3,x_4)$ with $x_1x_2x_3x_4\ne0$ is known on
\[
    a_1x_1^4+a_2x_2^4+a_3x_3^4+a_4x_4^4=0,
\]
    and $a_1a_2a_3a_4$ is a square in $\mathbb{Q}$, then another rational point is obtained unless the corresponding line lies on the surface.
\end{proposition}
