\begin{proposition}\label{eq:154}
    The equation
\[
    \begin{aligned}z^2&=U_1^2+U_2U_3,\\U_r&=a_rx^2+h_rxy+b_ry^2+f_ry+g_rx\end{aligned}
\]
    has an infinity of integer solutions if $h_r^2-4a_rb_r>0$ is not a perfect square for either $r=2$ or $r=3$, and either $U_2$ or $U_3$ is absolutely irreducible.
\end{proposition}
