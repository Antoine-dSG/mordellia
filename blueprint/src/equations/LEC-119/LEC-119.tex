\begin{proposition}
    Let
\[
    F=ax^2+by^2+cz^2,\qquad F_1=a_1x^2+b_1y^2+c_1z^2,\qquad G=px^2+qy^2+rz^2.
\]
    The equation
\[
    k_1FF_1=k_2G^2
\]
    has no nonzero integer solutions if the coefficients are integers, $k_1>0$, $k_2>0$, $k_1$ and $k_2$ have only divisors congruent to $1\pmod8$, and
\[
    a>0,\quad b>0,\quad c>0,\quad a\equiv b\equiv c\equiv-1\pmod8,
\]
    all odd divisors of
\[
    \Delta=
    \begin{vmatrix}
    a & b & c\\
    a_1 & b_1 & c_1\\
    p & q & r
    \end{vmatrix}
\]
    are congruent to $1\pmod8$, and either $\Delta$ is odd, or $\Delta$ is even and
\[
    a_1\equiv0\pmod2,\qquad b_1\equiv c_1\equiv1\pmod2,\qquad b_1+c_1\equiv4,6\pmod8.
\]
\end{proposition}
