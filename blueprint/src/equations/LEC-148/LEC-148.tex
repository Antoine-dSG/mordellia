\begin{proposition}
    All integer solutions of
\[
    x^4-2y^4=-z^2,\qquad x>0,\quad y>0,\quad z>0,\quad (x,y)=1,
\]
    are derived from the solution $(x,y,z)=(1,1,1)$.
\end{proposition}

\begin{proposition}
    The equation
\[
    x^4-2y^4=1
\]
    has only the integer solutions given by $y=0$, $x=\pm1$.
\end{proposition}
