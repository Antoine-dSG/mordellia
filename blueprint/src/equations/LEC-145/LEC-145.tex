\begin{proposition}
    Let $a,b,c,d$ be rational numbers with $acd\ne0$. If
\[
    (b^2-4ad)X^4-2bcX^2Y^2+c^2Y^4=Z^2
\]
    has only finitely many rational solutions, then
\[
    ax^4+bx^2y^2+dy^4=cz^2
\]
    has only finitely many rational solutions.
\end{proposition}

\begin{proposition}
    The equation
\[
    x^4+dy^4=z^2,\qquad (x,y)=1,\quad y\ne0,
\]
    has no integer solutions when $d=p$, $p\equiv7,11\pmod{16}$; when $d=2p$, $p\equiv\pm3\pmod8$; when $d=4p$, $p\equiv\pm3,-5\pmod{16}$; and when $d=-p$, $p\equiv\pm3,-5\pmod{16}$.
\end{proposition}
