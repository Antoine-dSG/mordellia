\begin{proposition}
    Over a quadratic field, the only non-rational solutions of
\[
    x^4+y^4=1
\]
    are
\[
    K=\mathbb{Q}(i),\qquad x=\pm i,\ y=0,\qquad x=0,\ y=\pm i.
\]
\end{proposition}

\begin{proposition}
    Over a cubic field, the only non-rational solutions, apart from rational ones, are
\[
    x=\pm i+ty,\qquad x^4+(\pm1+ty)^4=1,
\]
    or
\[
    y=\pm i+tx,\qquad y^4+(\pm1+tx)^4=1.
\]
\end{proposition}
