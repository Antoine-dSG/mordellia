\begin{proposition}
Let
\[
    L_r(\mathbf{x})=a_{r1}x_1+\cdots+a_{rn}x_n\quad(r=1,2,\ldots,n)
\]
be $n$ linear forms with integer coefficients and determinant $D>0$. Then the equations
\[
    L_r(\mathbf{x})=a_r\quad(r=1,2,\ldots,n)
\]
have an integer solution if and only if the congruences
\[
    L_r(\mathbf{x})\equiv a_r\pmod M\quad(r=1,2,\ldots,n)
\]
are solvable for all $M$.
\end{proposition}
