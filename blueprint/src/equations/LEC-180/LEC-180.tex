\begin{proposition}
    Let $K$ be a primitive algebraic number field of degree $n$, and let $r=r_1+r_2-1$, where $r_1$ is the number of real conjugate fields and $r_2$ is the number of pairs of conjugate complex fields. If $r<n-3$ and $\alpha,\beta,\gamma$ are three linearly independent integers in $K$, then
\[
    N(\alpha x+\beta y+\gamma z)=A
\]
    has only finitely many integer solutions.
\end{proposition}
